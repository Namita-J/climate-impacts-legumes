# 4. Bootstrap regression function (fit model to resampled studies)
boot_fun <- function(d, i) {
  d2 <- d[i, ]
  fit <- lm(dp ~ poly(dt, 2, raw = TRUE) * adaptation, data = d2)
  coef(fit)
}

# 5. Run bootstrap
set.seed(123)
boot_out <- boot(dat, boot_fun, R = 500, strata = dat$study_id)

# 6. Make prediction grid
grid_df <- expand.grid(
  dt = seq(min(dat$dt), max(dat$dt), length.out = 100),
  adaptation = levels(dat$adaptation)
)

# 7. Create design matrix using same formula as in bootstrap
X <- model.matrix(~ poly(dt, 2, raw = TRUE) * adaptation, data = grid_df)

# 8. Multiply to get predictions for each bootstrap draw
#    Each column of pred_mat is predictions from one bootstrap model
pred_mat <- X %*% t(boot_out$t)

# 9. Summarize to CI bands
grid_df <- grid_df %>%
  mutate(
    fit  = rowMeans(pred_mat, na.rm = TRUE),
    lo95 = apply(pred_mat, 1, quantile, probs = 0.025, na.rm = TRUE),
    hi95 = apply(pred_mat, 1, quantile, probs = 0.975, na.rm = TRUE)
  )

  coef(fit)

# 5. Function to run bootstrapping (remains the same, but will be applied to more groups)
get_grid <- function(subdat, group_name) {
  if (nrow(subdat) == 0) {
    warning(paste("No data available for the group:", group_name, ". Skipping bootstrap."), call. = FALSE)
    return(NULL)
  }
  set.seed(123)
  boot_out <- boot(subdat, boot_fun, R = 1000, strata = subdat$study_id)
  grid <- expand.grid(
    dt = seq(min(subdat$dt, na.rm = TRUE), max(subdat$dt, na.rm = TRUE), length.out = 100),
    adaptation = levels(subdat$adaptation)
  )
  X <- model.matrix(~ poly(dt, 2, raw = TRUE) * adaptation, data = grid)
  pred_mat <- X %*% t(boot_out$t)
  grid %>%
    mutate(
      fit = rowMeans(pred_mat, na.rm = TRUE),
      lo95 = apply(pred_mat, 1, quantile, probs = 0.025, na.rm = TRUE),
      hi95 = apply(pred_mat, 1, quantile, probs = 0.900, na.rm = TRUE)
    )
}

  select(crop_type, co2_group, grid_data) %>%
  # Unnest the results to create a single large dataframe

  coef(fit)

  adaptation = levels(dat$adaptation)
)

# 7. Create design matrix using same formula as in bootstrap
X <- model.matrix(~ poly(dprec, 2, raw = TRUE) * adaptation, data = grid_df)

# 8. Multiply to get predictions for each bootstrap draw
#    Each column of pred_mat is predictions from one bootstrap model
pred_mat <- X %*% t(boot_out$t)

# 9. Summarize to CI bands
grid_df <- grid_df %>%
  mutate(
    fit  = rowMeans(pred_mat, na.rm = TRUE),
    lo95 = apply(pred_mat, 1, quantile, probs = 0.025, na.rm = TRUE),
    hi95 = apply(pred_mat, 1, quantile, probs = 0.975, na.rm = TRUE)
  )

# 6. Bootstrap function -----------------------------------------------------
boot_ci <- function(d) {
  boot_mean <- function(data, i) mean(data[i], na.rm = TRUE)
  bt <- boot(d, boot_mean, R = 1000)
  ci <- boot.ci(bt, type = "perc")
  tibble(
    mean = mean(d, na.rm = TRUE),
    lower = ci$percent[4],
    upper = ci$percent[5]
  )
}

plot_bi0 <- plot_df %>%
  dplyr::mutate(
    certainty_raw = -ci_range,  # narrower CI => larger value
    certainty = if (all(is.na(certainty_raw))) NA_real_ else
      scales::rescale(certainty_raw, to = c(0, 1),
                      from = range(certainty_raw, na.rm = TRUE))
  )

non_na <- plot_bi0 %>%
  dplyr::filter(!is.na(mean_protein_yield), !is.na(certainty))

non_na_bi <- biscale::bi_class(non_na,
                               x = mean_protein_yield,
                               y = certainty,
                               style = "quantile", dim = 3)

plot_ok  <- non_na_bi
plot_nas <- plot_bi0 %>%
  dplyr::filter(is.na(mean_protein_yield) | is.na(certainty))

# 2) Main bivariate heatmap (no legend here)
p_main <- ggplot() +
  # NA tiles
  geom_tile(
    data = plot_nas,
    aes(x = period, y = crop),
    fill = "grey90", color = "black", linewidth = 0.5,
    width = 0.9, height = 0.9
  ) +
  # bivariate tiles
  geom_tile(
    data = plot_ok,
    aes(x = period, y = crop, fill = bi_class),
    color = "black", linewidth = 0.5,
    width = 0.9, height = 0.9
  ) +
  # labels
  geom_text(
    data = plot_bi0,
    aes(x = period, y = crop, label = label),
    size = 4.5, lineheight = 0.95
  ) +
  # palette
  bi_scale_fill(pal = "BlueYI", dim = 3) +
  labs(
    title = "Projected Change in Digestible Protein Yield (t/ha)",
    subtitle = "Bivariate tiles: hue = mean Δ protein yield; lightness = certainty (narrower CI = darker)",
    x = NULL, y = NULL
  ) +
  facet_wrap(~ panel, nrow = 1) +
  theme_minimal(base_size = 16) +
  theme(
    legend.position = "none",  # legend drawn separately
    strip.text   = element_text(size = 14, face = "bold"),
    plot.title   = element_text(size = 18, face = "bold"),
    plot.subtitle= element_text(size = 13),
    axis.text.x  = element_text(size = 12),
    axis.text.y  = element_text(size = 12),
    panel.spacing = unit(1, "lines"),
    panel.grid = element_blank(),
    plot.margin = margin(10, 90, 10, 10),                # <-- right margin for legend
    plot.background = element_rect(fill = "white", color = NA)
  )

# 3) Compact bivariate legend (character labels only)
leg <- biscale::bi_legend(
  pal  = "BlueYI",
  dim  = 3,
  xlab = "Δ Protein yield (low → high)",
  ylab = "Certainty (low → high)",
  size = 10
)

# --- assemble figure: dedicate a column to the legend (no overlap) ---
final_plot <- cowplot::plot_grid(
  p_main + theme(plot.margin = margin(10, 10, 10, 10)),
  leg    + theme(plot.margin = margin(20, 10, 20, 10)),
  ncol = 2,
  rel_widths = c(1, 0.22),   # <- adjust legend column width here
  align = "h"
)

print(final_plot)
