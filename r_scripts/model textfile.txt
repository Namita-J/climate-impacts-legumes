# --- 1. Data Import & Cleaning ---
raw <- read_excel("excel_data/yield_predictions_data_with_nextgddp_yearly_baseline_deltas.xlsx", sheet = "Sheet1") %>%
  janitor::clean_names()

# --- 2. Define Legume Crops ---
legume_crops <- c("Bambara", "Bush Bean", "Chickpea", "Climbing Bean", "Common Bean", 
                  "Cowpea", "Faba Bean", "Green grams", "Groundnut", "Soybean")

# --- 3. Filter and Recode ---
df_legumes <- raw %>%
  filter(crop %in% legume_crops) %>%
  mutate(
    dp         = as.numeric(climate_impacts_percent),
    dt         = as.numeric(temp_change_nextgddp),
    precip     = as.numeric(precip_change_nextgddp),
    adaptation = factor(if_else(adaptation_type == "No", "No-adaptation", "Adaptation"),
                        levels = c("No-adaptation", "Adaptation")),
    co2_group  = factor(if_else(co2 == "Yes", "With CO2 fertilization", "Without CO2 fertilization"),
                        levels = c("Without CO2 fertilization", "With CO2 fertilization")),
    study_id   = factor(ref_no),
    crop       = factor(crop),
    country    = factor(country)
  ) %>%
  filter(complete.cases(dp, dt, precip, study_id, crop, country, adaptation, co2_group))

# --- 4. LINEAR MODELS ---
main_lm <- lm(dp ~ dt + precip + co2_group + adaptation, data = df_legumes)
full_lm <- lm(dp ~ (dt + precip + co2_group + adaptation)^2, data = df_legumes)

# --- 5. Clustered SE ---
main_clustered <- coeftest(main_lm, vcov = vcovCL(main_lm, cluster = df_legumes$study_id))
full_clustered <- coeftest(full_lm, vcov = vcovCL(full_lm, cluster = df_legumes$study_id))

# --- 6. VIF ---
vif(main_lm)

# --- 7. Residuals ---
par(mfrow = c(1,2))
plot(main_lm, which = 1)
plot(main_lm, which = 2)

# --- 8. GAM Test for Linearity ---
gam_test <- gam(dp ~ s(precip) + s(dt) + adaptation + co2_group, 
                data = df_legumes, method = "REML")

par(mfrow = c(1, 2))
plot(gam_test, residuals = TRUE, shade = TRUE)
summary(gam_test)

# Combine GAM parametric and smooth terms into one clean summary
gamm_summary_tbl <- bind_rows(
  gam_param %>%
    select(term, estimate, std.error, statistic, p.value) %>%
    mutate(component = "Parametric"),
  gam_smooth %>%
    rename(term = smooth, statistic = statistic, p.value = p.value) %>%
    mutate(
      estimate = NA,
      std.error = NA,
      component = "Smooth"
    ) %>%
    select(term, estimate, std.error, statistic, p.value, component)
)
